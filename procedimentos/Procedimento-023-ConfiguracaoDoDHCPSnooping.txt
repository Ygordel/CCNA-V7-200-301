!======================================================================
! 01 - CONCEITO
!======================================================================

! DHCP Snooping é um recurso de segurança dos Switches Cisco utilizado
! para controlar e validar mensagens DHCP na rede.

! O recurso ajuda a mitigar ataques relacionados ao DHCP, como a
! utilização de servidores DHCP não autorizados.

! Quando habilitado, as interfaces são consideradas não confiáveis
! (Untrusted) por padrão.

! As interfaces conectadas a servidores DHCP ou equipamentos que
! fornecem o serviço DHCP devem ser configuradas como confiáveis
! (Trusted).


!======================================================================
! 02 - DHCP SNOOPING
!======================================================================

! O DHCP Snooping mantém informações relacionadas às associações
! entre:

! - Endereço IP;
! - Endereço MAC;
! - Interface;
! - VLAN;
! - Tempo de concessão (Lease Time).

! Essas informações podem ser consultadas por meio da tabela
! de associações do DHCP Snooping.


!======================================================================
! 03 - SWITCH LAYER 3 3560
!======================================================================

enable
configure terminal


! Habilitando o DHCP Snooping:

ip dhcp snooping


! Definindo a VLAN monitorada:

ip dhcp snooping vlan 1


! Habilitando a Information Option:

ip dhcp snooping information option


! Permitindo Information Option em interfaces não confiáveis:

ip dhcp snooping information option allow-untrusted


!======================================================================
! 04 - PORTA DO SERVIDOR DHCP
!======================================================================

! A porta conectada ao servidor DHCP deve ser considerada confiável.

interface fastEthernet 0/1

ip dhcp snooping trust

exit


!======================================================================
! 05 - PORTA DO ACCESS POINT
!======================================================================

! É possível limitar a quantidade de solicitações DHCP recebidas
! em uma determinada interface.

interface fastEthernet 0/23

ip dhcp snooping limit rate 10

end


! Salvando a configuração:

write


!======================================================================
! 06 - SWITCH LAYER 2 2960
!======================================================================

enable
configure terminal


! Habilitando o DHCP Snooping:

ip dhcp snooping


! Definindo a VLAN monitorada:

ip dhcp snooping vlan 1


! Habilitando a Information Option:

ip dhcp snooping information option


! Permitindo Information Option:

ip dhcp snooping information option allow-untrusted


!======================================================================
! 07 - INTERFACES DE TRUNK
!======================================================================

! As interfaces de trunk utilizadas para comunicação com outros
! equipamentos da infraestrutura devem ser configuradas como Trusted.

interface range gigabitEthernet 0/1 - 2

ip dhcp snooping trust

exit


!======================================================================
! 08 - INTERFACES DE ACCESS
!======================================================================

! Limitando a quantidade de requisições DHCP nas portas dos
! dispositivos finais.

interface range fastEthernet 0/1 - 2

ip dhcp snooping limit rate 5

end


! Salvando a configuração:

write


!======================================================================
! 09 - VERIFICAÇÃO DA CONFIGURAÇÃO
!======================================================================

! Visualizando as configurações relacionadas ao DHCP Snooping:

show running-config | section ip dhcp


! Visualizando as configurações das interfaces:

show running-config | section interface


! Visualizando o estado do DHCP Snooping:

show ip dhcp snooping


! Visualizando a tabela de associações:

show ip dhcp snooping binding


! Visualizando informações do banco de dados:

show ip dhcp snooping database


!======================================================================
! 10 - TESTE NOS CLIENTES
!======================================================================

! No Microsoft Windows, liberar o endereço IPv4 atual:

ipconfig /release


! Solicitar um novo endereço IPv4 ao servidor DHCP:

ipconfig /renew


! Após a renovação, verificar a associação criada pelo DHCP Snooping:

show ip dhcp snooping binding
