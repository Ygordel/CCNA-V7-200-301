!======================================================================
! 02 - CONFIGURAÇÃO BASE DO SWITCH LAYER 2 2960-5
!======================================================================

enable


! Configurando data e hora:

clock set 14:00:00 26 May 2020


! Configurando histórico de comandos:

terminal history size 256


! Habilitando monitoramento de mensagens no terminal:

terminal monitor


configure terminal


! Configurando o hostname:

hostname sw-l2-2960-5


! Criptografando senhas:

service password-encryption


! Configurando marcação de data e hora nos Logs:

service timestamps log datetime msec


! Habilitando o recurso de Logging:

logging on


! Configurando o servidor Syslog:

logging host 192.168.3.1


! Configurando o nível de mensagens:

logging trap debugging


! Configurando comunidade SNMP somente leitura:

snmp-server community ptiro ro


! Configurando comunidade SNMP leitura e escrita:

snmp-server community ptirw rw


! Habilitando CDP:

cdp run


! Habilitando LLDP:

lldp run


! Habilitando resolução de nomes:

ip domain-lookup


! Configurando o domínio:

ip domain-name vaamonde.pti


! Configurando o servidor DNS:

ip name-server 192.168.3.1


! Configurando o Gateway padrão:

ip default-gateway 192.168.99.254


! Configurando o fuso horário:

clock timezone BR -3


! Habilitando autenticação NTP:

ntp authenticate


! Configurando chave de autenticação NTP:

ntp authentication-key 123456 md5 vaamonde


! Configurando servidor NTP:

ntp server 192.168.3.1


! Configurando mensagem de acesso:

banner motd #AVISO: acesso autorizado somente a funcionarios#


! Configurando senha do modo privilegiado:

enable secret vaamonde@pti


! Criando usuário local:

username robson secret vaamonde@pti

username vaamonde password vaamonde@pti

username admin privilege 15 secret vaamonde@pti


! Gerando chaves RSA para SSH:

crypto key generate rsa usage-keys modulus 1024


! Configurando SSH versão 2:

ip ssh version 2


! Configurando tempo limite do SSH:

ip ssh time-out 60


! Configurando número de tentativas de autenticação:

ip ssh authentication-retries 2


! Configurando acesso ao Console:

line console 0

login local

password vaamonde@pti

logging synchronous

exec-timeout 5 30

exit


! Configurando acesso remoto VTY:

line vty 0 4

login local

password vaamonde@pti

logging synchronous

exec-timeout 5 30

transport input ssh

end


! Salvando as configurações:

write
